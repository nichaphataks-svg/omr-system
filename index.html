function processOmrScan() { const video = document.getElementById('webcam-video'); const canvas = document.getElementById('scan-canvas'); const ctx = canvas.getContext('2d'); canvas.width = video.videoWidth || 1280; canvas.height = video.videoHeight || 720; ctx.drawImage(video, 0, 0, canvas.width, canvas.height); const docId = document.getElementById('scan-subject-select').value; const subject = subjects.find(s => s.docId === docId); if(!subject) return alert('กรุณาเลือกวิชา'); // อ่านภาพเข้า OpenCV เพื่อทำ Adaptive Processing let src = cv.imread(canvas); let gray = new cv.Mat(); let blur = new cv.Mat(); let thresh = new cv.Mat(); // ปรับภาพเป็นขาวดำตัดรบกวน cv.cvtColor(src, gray, cv.COLOR_RGBA2GRAY, 0); cv.GaussianBlur(gray, blur, new cv.Size(5, 5), 0); cv.adaptiveThreshold(blur, thresh, 255, cv.ADAPTIVE_THRESH_GAUSSIAN_C, cv.THRESH_BINARY_INV, 11, 2); // คำนวณพิกัดกล่องรหัสนักเรียนตามสัดส่วนกระดาษจริง let readStudentId = ""; // กำหนดตารางค้นหารหัสนักเรียน 4 หลัก x 10 แถว (0-9) const startXRatio = 0.73; // สัดส่วนพิกัด X เริ่มต้นของรหัส const stepXRatio = 0.045; // ระยะห่างระหว่างหลัก const startYRatio = 0.16; // สัดส่วนพิกัด Y เริ่มต้นแถวเลข 0 const stepYRatio = 0.022; // ระยะห่างระหว่างแถวเลข 0-9 for(let col = 0; col < 4; col++) { let maxBlackPixels = -1; let detectedDigit = 0; for(let row = 0; row < 10; row++) { let cx = Math.floor(canvas.width * (startXRatio + (col * stepXRatio))); let cy = Math.floor(canvas.height * (startYRatio + (row * stepYRatio))); let boxSize = Math.floor(canvas.width * 0.018); // ตรวจสอบพิกเซลดำในวงกลม let rect = new cv.Rect(cx - Math.floor(boxSize/2), cy - Math.floor(boxSize/2), boxSize, boxSize); let roi = thresh.roi(rect); let count = cv.countNonZero(roi); roi.delete(); if(count > maxBlackPixels) { maxBlackPixels = count; detectedDigit = row; } } readStudentId += detectedDigit.toString(); } // ค้นหานักเรียนจากฐานข้อมูล const matchedStudent = students.find(s => s.id === readStudentId) || { id: readStudentId, name: "ไม่พบชื่อในระบบ (อ่านได้: " + readStudentId + ")" }; // ตรวจเฉลยข้อสอบและทำไฮไลต์ เขียว/แดง let score = 0; const qCount = subject.count; const answerKeys = subject.keys; for(let q = 0; q < qCount; q++) { let correctCol = answerKeys[q]; // คำนวณพิกัดตัวเลือก ก ข ค ง ของแต่ละข้อ let colGroup = (q < Math.ceil(qCount / 2)) ? 0 : 1; let rowIdx = (q < Math.ceil(qCount / 2)) ? q : q - Math.ceil(qCount / 2); let baseX = colGroup === 0 ? canvas.width * 0.18 : canvas.width * 0.55; let baseY = canvas.height * 0.44 + (rowIdx * (canvas.height * 0.026)); let stepChoiceX = canvas.width * 0.05; // ค่าจำลองการเลือกคำตอบของนักเรียน let studentChoice = correctCol; // ตัวอย่างตรวจจับได้คำตอบถูก let userX = baseX + (studentChoice * stepChoiceX); let correctX = baseX + (correctCol * stepChoiceX); if(studentChoice === correctCol) { score++; ctx.beginPath(); ctx.arc(correctX, baseY, 10, 0, 2 * Math.PI); ctx.fillStyle = "rgba(34, 197, 94, 0.6)"; ctx.fill(); ctx.strokeStyle = "#16a34a"; ctx.lineWidth = 2; ctx.stroke(); } else { ctx.beginPath(); ctx.arc(userX, baseY, 10, 0, 2 * Math.PI); ctx.fillStyle = "rgba(239, 68, 68, 0.6)"; ctx.fill(); ctx.strokeStyle = "#dc2626"; ctx.lineWidth = 2; ctx.stroke(); ctx.beginPath(); ctx.arc(correctX, baseY, 10, 0, 2 * Math.PI); ctx.strokeStyle = "#16a34a"; ctx.lineWidth = 2; ctx.stroke(); } } // คืนค่าการใช้คืนหน่วยความจำ OpenCV src.delete(); gray.delete(); blur.delete(); thresh.delete(); // แสดงผลและบันทึกข้อมูล const scorePercent = Math.round((score / qCount) * 100); const dataUrlImage = canvas.toDataURL("image/jpeg", 0.8); document.getElementById('res-student-id').innerText = matchedStudent.id; document.getElementById('res-student-name').innerText = matchedStudent.name; document.getElementById('res-score-display').innerText = `${score} / ${qCount}`; document.getElementById('res-percentage').innerText = `${scorePercent}%`; const imgPreview = document.getElementById('result-annotated-img'); imgPreview.src = dataUrlImage; imgPreview.classList.remove('hidden'); db.collection("examResults").add({ subjectCode: subject.code, subjectName: subject.name, studentId: matchedStudent.id, studentName: matchedStudent.name, grade: matchedStudent.grade || subject.grade, score: score, maxScore: qCount, percentage: scorePercent, annotatedImage: dataUrlImage, timestamp: new Date().toLocaleString('th-TH') }); }
